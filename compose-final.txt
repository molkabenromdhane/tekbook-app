name: tekbook-app
services:
  db:
    environment:
      POSTGRES_DB: tekbook
      POSTGRES_PASSWORD: Dem0-Lab-Only
      POSTGRES_USER: tekbook
    healthcheck:
      test:
        - CMD-SHELL
        - pg_isready -U tekbook -d tekbook
      timeout: 3s
      interval: 5s
      retries: 5
    image: postgres:16
    networks:
      backend: null
    volumes:
      - type: volume
        source: db-data
        target: /var/lib/postgresql/data
        volume: {}
  tekbook:
    build:
      context: /home/molka/tekbook-app
      dockerfile: Dockerfile
    depends_on:
      db:
        condition: service_healthy
        required: true
    environment:
      SPRING_DATASOURCE_PASSWORD: Dem0-Lab-Only
      SPRING_DATASOURCE_URL: jdbc:postgresql://db:5432/tekbook
      SPRING_DATASOURCE_USERNAME: tekbook
    networks:
      backend: null
      frontend: null
    ports:
      - mode: ingress
        target: 8080
        published: "8080"
        protocol: tcp
    restart: unless-stopped
networks:
  backend:
    name: tekbook-app_backend
  frontend:
    name: tekbook-app_frontend
volumes:
  db-data:
    name: tekbook-app_db-data
